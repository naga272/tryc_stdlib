/*
* Macro con parametri (stile C) e macro multi-riga.
*/

#define SQUARE(x) ((x) * (x))
#define ADD(a, b) ((a) + (b))

// una macro con parametri puo' richiamarne un'altra nel proprio corpo
#define SUM_OF_SQUARES(a, b) (SQUARE(a) + SQUARE(b))

// macro multi-riga: '\' a fine riga continua la #define sulla riga successiva
#define GREET(name) \
    print("Hello, " + name + "!\n")

val1 = SQUARE(5)
print("SQUARE(5) = " + itoa(val1) + "\n")

val2 = ADD(2, 3)
print("ADD(2, 3) = " + itoa(val2) + "\n")

val3 = SUM_OF_SQUARES(2, 3)
print("SUM_OF_SQUARES(2, 3) = " + itoa(val3) + "\n")

GREET("world")
