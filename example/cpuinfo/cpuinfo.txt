print("               CPU INFO\n\n\n")
print("Vendor:  " + cpu_vendor() + "\n")

print("Model:   " + itoa(cpu_model()) + "\n")
print("Stepping: " + itoa(cpu_stepping()) + "\n")

print("RDRAND:  " + itoa(has_rdrand()) + "\n")
