*   Standard library per ottenere informazioni sulla CPU, tramite
*   l'istruzione CPUID (identificazione/funzionalita' hardware) e la
*   syscall sched_getaffinity (quante CPU logiche sono disponibili al
*   processo corrente).
*   Non viene usata da nessuna parte l'heap memory
*/

#define SYS_SCHED_GETAFFINITY 204
#endif

class CpuidResult --no-padding:
    eax int
    ebx int
    ecx int
    edx int


def cpuid(leaf: int, subleaf: int, out: CpuidResult) -> None:
    /*
    * Esegue l'istruzione CPUID(leaf, subleaf) e scrive i 4 registri
    * risultato (EAX, EBX, ECX, EDX) nei campi di out
    * argomenti:
    *   leaf: valore da caricare in EAX prima di CPUID
    *   subleaf: valore da caricare in ECX prima di CPUID (ignorato
    *     dalla maggior parte dei leaf - passa 0 se non sai se serve)
    *   out: struttura CpuidResult in cui scrivere EAX/EBX/ECX/EDX
    * return: nessuno (procedura)
    */
    asminline(
        ; callee-saved
        push rbx
        mov r8, rdx     ; out* (rdx verra' distrutto da cpuid)
        mov eax, edi    ; leaf
        mov ecx, esi    ; subleaf
        cpuid
        mov [r8], eax
        mov [r8 + 4], ebx
        mov [r8 + 8], ecx
        mov [r8 + 12], edx
        pop rbx
    ) endasm


def cpu_vendor() -> str:
    /*
    * Stringa di 12 caratteri che identifica il produttore della CPU
    * (es. "GenuineIntel", "AuthenticAMD") CPUID leaf 0.
    * return: stringa di 12 caratteri
    */
    asminline(
        push rbx
        mov rdi, 13     ; + terminatore
        call rt_alloc
        mov r8, rax

        mov eax, CPUID_LEAF_VENDOR
        xor ecx, ecx
        cpuid

        mov [r8], ebx
        mov [r8 + 4], edx
        mov [r8 + 8], ecx

        mov rax, r8

        push rax
        mov rdi, rax
        mov rsi, 1
        call rt_promote
        pop rax

        pop rbx
    ) endasm


def cpu_brand() -> str:
    /*
    * Stringa descrittiva completa della CPU, es.
    * "Intel(R) Core(TM) i7-9700K CPU @ 3.60GHz"
    * CPUID leaf 0x80000002/3/4 (48 byte totali, 16 per leaf).
    * NB: puo' contenere spazi finali di padding
    */
    asminline(
        push rbx
        mov rdi, 49              ; 48 caratteri + terminatore
        call rt_alloc
        mov r8, rax

        mov eax, CPUID_LEAF_BRAND_1
        xor ecx, ecx
        cpuid
        mov [r8], eax
        mov [r8 + 4], ebx
        mov [r8 + 8], ecx
        mov [r8 + 12], edx

        mov eax, CPUID_LEAF_BRAND_2
        xor ecx, ecx
        cpuid
        mov [r8 + 16], eax
        mov [r8 + 20], ebx
        mov [r8 + 24], ecx
        mov [r8 + 28], edx

        mov eax, CPUID_LEAF_BRAND_3
        xor ecx, ecx
        cpuid
        mov [r8 + 32], eax
        mov [r8 + 36], ebx
        mov [r8 + 40], ecx
        mov [r8 + 44], edx

        mov rax, r8

        push rax
        mov rdi, rax
        mov rsi, 1
        call rt_promote
        pop rax

        pop rbx
    ) endasm


def cpu_family() -> int:
    /*
    * extended Family della CPU (CPUID leaf 1, EAX)
    * return: family della CPU
    */
    asminline(
        push rbx
        mov eax, CPUID_LEAF_FEATURES
        xor ecx, ecx
        cpuid

        mov r8d, eax
        shr r8d, 8
        and r8d, 0xF              ; family di base

        cmp r8d, 0xF
        jne .done
            mov r9d, eax
            shr r9d, 20
            and r9d, 0xFF          ; extended family
            add r8d, r9d
        .done:

        mov rax, r8
        pop rbx
    ) endasm


def cpu_model() -> int:
    /*
    * extended model della CPU (CPUID leaf 1, EAX)
    * return: model della CPU
    */
    asminline(
        push rbx
        mov eax, CPUID_LEAF_FEATURES
        xor ecx, ecx
        cpuid

        mov r8d, eax
        shr r8d, 8
        and r8d, 0xF               ; family di base (decide se serve extended model)

        mov r9d, eax
        shr r9d, 4
        and r9d, 0xF                ; model di base

        cmp r8d, 6
        je .use_extended
        cmp r8d, 0xF
        je .use_extended
        jmp .done

        .use_extended:
            mov r10d, eax
            shr r10d, 16
            and r10d, 0xF             ; extended model
            shl r10d, 4
            or r9d, r10d
        .done:

        mov rax, r9
        pop rbx
    ) endasm


def cpu_stepping() -> int:
    /*
    * Stepping della CPU (CPUID leaf 1, EAX bit 3:0)
    * return: stepping della CPU
    */
    asminline(
        push rbx
        mov eax, CPUID_LEAF_FEATURES
        xor ecx, ecx
        cpuid
        mov r8d, eax
        and r8d, 0xF
        mov rax, r8
        pop rbx
    ) endasm


def cpu_cache_line_size() -> int:
    /*
    * Numero di bytes di una linea di cache
    * (CPUID leaf 1, EBX bit 15:8, espressa nel registro in unita' da 8 byte).
    * return: dimensione di una linea di cache in byte
    */
    asminline(
        push rbx
        mov eax, CPUID_LEAF_FEATURES
        xor ecx, ecx
        cpuid
        mov r8d, ebx
        shr r8d, 8
        and r8d, 0xFF
        shl r8d, 3      ; * 8
        mov rax, r8
        pop rbx
    ) endasm


def physical_address_bits() -> int:
    /*
    * Numero di bit usati dalla CPU per indirizzare la memoria fisica
    * (CPUID leaf 0x80000008, EAX bit 7:0) (determina quanta RAM la
    * CPU puo' indirizzare in teoria (2^n byte)).
    * return: numero di bit di indirizzamento fisico
    */
    asminline(
        push rbx
        mov eax, CPUID_LEAF_ADDRESS_SIZES
        xor ecx, ecx
        cpuid
        mov r8d, eax
        and r8d, 0xFF
        mov rax, r8
        pop rbx
    ) endasm

def logical_cpu_count() -> int:
    /*
    * return: numero di CPU logiche disponibili a questo processo
    */
    asminline(
        push rbx
        mov rdi, 128
        call rt_alloc
        mov rbx, rax
        ; pid = 0 -> processo corrente
        xor rdi, rdi
        ; cpusetsize
        mov rsi, 128
        ; mask*
        mov rdx, rbx
        mov rax, SYS_SCHED_GETAFFINITY
        syscall

        ; conta i bit a 1 nei 128 byte del buffer
        xor r8, r8      ; r8 = total counter
        xor r9, r9      ; r9 = idx

        .byte_loop:
            cmp r9, 128
            jge .done

            movzx r10, byte [rbx + r9]
            .count_bits:
                test r10, r10
                jz .next_byte
                mov r11, r10
                dec r11
                and r10, r11
                inc r8
                jmp .count_bits
            .next_byte:
            inc r9
            jmp .byte_loop

        .done:
        mov rax, r8
        pop rbx
    ) endasm


def has_sse2() -> int:
    /*
    * Se la CPU supporta SSE2 (CPUID leaf 1, EDX bit 26)
    * return: 1 se supportato, 0 altrimenti
    */
    asminline(
        push rbx
        mov eax, CPUID_LEAF_FEATURES
        xor ecx, ecx
        cpuid
        and edx, CPUID_EDX_SSE2
        cmp edx, 0
        setne al
        movzx rax, al
        pop rbx
    ) endasm
