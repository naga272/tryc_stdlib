def sqlite3_checker_error(dbs: sqlite3, response: int, error: str) -> int:
    if response != SQLITE_OK:
        print(error + "\n")
        sqlite3_free(error)
        sqlite3_close(dbs)
        exit(1)

    sqlite3_free(error)
    return response

    return 0


def sqlite3_exec_w(dbs: sqlite3, query: str) -> int:
    errmsg =: str
    response = sqlite3_exec(dbs, query, 0, 0, &errmsg)
    response = sqlite3_checker_error(dbs, response, errmsg)
    return response


def sqlite3_drop_table(dbs: sqlite3, name_table: str) -> int:
    errmsg =: str
    query = "DROP TABLE " + name_table + ";"
    response = sqlite3_exec(dbs, query, 0, 0, &errmsg)
    return sqlite3_checker_error(dbs, response, errmsg)


def sqlite3_run_query(dbs: sqlite3) -> None:
    err_msg =: str
    response = sqlite3_exec(dbs, "SELECT * FROM Users;", 0, 0, &err_msg)
    sqlite3_checker_error(dbs, response, err_msg)


def sqlite3_prepare_stmnt(dbs: sqlite3, sql: str, stmt: &sqlite3_stmt) -> int:
    response = sqlite3_prepare_v2(dbs, sql, -1, stmt, 0)
    return response


def sqlite3_bind(stmt: sqlite3_stmt, index: int, text: str) -> int:
    size_text = strlen(text)
    response = sqlite3_bind_text(stmt, index, text, size_text, SQLITE_TRANSIENT)
    return response
