def sys_fork() -> int:
    asminline(
        mov rax, SYS_FORK
        syscall
    ) endasm


def sys_kill() -> None:
    asminline(
        mov rax, SYS_KILL
        syscall
    ) endasm


def sys_vfork() -> int:
    asminline(
        mov rax, SYS_VFORK
        syscall
    )endasm


def fork() -> int:
    /*
    * val < 0: The creation of a child process was unsuccessful, then it returns -1.
    * val == Zero: Returned to the newly created child process.
    * val > 0: Returned to parent or caller. The value contains the process ID of the newly created child process.
    */
    pid = sys_fork()
    if pid < 0:
        print("error, pid value: " + itoa(pid) + "\n")
    return pid

class ArgsClone:
    reg_rdx str
    reg_rdi str
    reg_rsi str

    reg_r10 str
    reg_r11 str
    reg_r12 str

def sys_clone(stack_child_ptr: str) -> int:
    asminline(
        mov rsi, rdi
        mov rdi, CLONE_VM | CLONE_SIGHAND | CLONE_FILES | CLONE_FS | SIGCHLD
        xor rdx, rdx 
        xor r10, r10
        xor r8, r8
        xor r9, r9
        mov rax, SYS_CLONE
        syscall
    )endasm


def do_clone(function: str, args: &ArgsClone) -> None:
    stack_child_ptr = malloc(CHILD_STACK_SIZE)
    asminline (
        ; dall'alto verso il basso
        add qword[rbp - 64], CHILD_STACK_SIZE
        ; allineamento stack child
        and qword[rbp - 64], -16
    ) endasm
    
    response = sys_clone(stack_child_ptr)
    if response < 0:
        print("Errore nel tentativo di fare la clone()\n")
        exit(0)
    
    // caso child
    if response == 0:
        asminline (
            ; ripristino i registri
            mov     rbx, qword[rbp - 56]
            mov     rdx, qword[rbx + 0]
            mov     rdi, qword[rbx + 8]
            mov     rsi, qword[rbx + 16]
            mov     r8,  qword[rbx + 24]
            mov     r9,  qword[rbx + 32]
            mov     r10, qword[rbx + 40]
            mov     r11, qword[rbx + 48]
            mov     r12, qword[rbx + 56]
            mov     r13, qword[rbx + 64]
            mov     r14, qword[rbx + 72]
            mov     r15, qword[rbx + 80]
            ; chiamo tramite ptr a funzione
            push rdi
            mov rdi, rbx
            call tryc_free
            pop rdi
            call [rbp - 48]
        ) endasm
        exit(0)
    free(stack_child_ptr)

def sys_pid() -> int:
    /*
    * PID (process id) del processo corrente
    */
    asminline(
        mov rax, SYS_GETPID
        syscall
    ) endasm


def sys_ppid() -> int:
    /*
    * PID del processo padre
    */
    asminline(
        mov rax, SYS_GETPPID
        syscall
    ) endasm


def sys_uid() -> int:
    /*
    * User id del processo (chi ha lanciato il programma)
    */
    asminline(
        mov rax, SYS_GETUID
        syscall
    ) endasm


def sys_euid() -> int:
    /*
    * User id effettivo (puo' differire dal reale se il
    * binario ha il bit setuid attivo)
    */
    asminline(
        mov rax, SYS_GETEUID
        syscall
    ) endasm


def sys_gid() -> int:

    * Group id reale del processo
    */
    asminline(
        mov rax, SYS_GETGID
        syscall
    ) endasm


def sys_egid() -> int:
    /*
    * Group id effettivo del processo (puo' differire dal reale se il
    * binario ha il bit setgid attivo)
    */
    asminline(
        mov rax, SYS_GETEGID
        syscall
    ) endasm
